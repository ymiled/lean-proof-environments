Context already in scope:

```lean
inductive Trellis where
  | stem (n : Nat)
  | brim (x : Nat)
  | seed (a b : Trellis)

inductive Vane where
  | cusp
  | root (x : Nat) (e : Trellis)
  | flint (c d : Vane)
  | base (e : Trellis) (c d : Vane)

abbrev Warp := Nat → Nat
abbrev Strand := Nat → Bool   -- true = secret, false = public

def clasp (s : Warp) (x v : Nat) : Warp := fun y => if y = x then v else s y

def weave (s : Warp) : Trellis → Nat
  | .stem n    => n
  | .brim x    => s x
  | .seed a b => weave s a + weave s b

def fuse (G : Strand) : Trellis → Bool
  | .stem _    => false
  | .brim x    => G x
  | .seed a b => fuse G a || fuse G b

def hew (s : Warp) : Vane → Warp
  | .cusp       => s
  | .root x e => clasp s x (weave s e)
  | .flint c d    => hew (hew s c) d
  | .base e c d  => if weave s e = 0 then hew s d else hew s c

def graft (G : Strand) (pc : Bool) : Vane → Bool
  | .cusp       => true
  | .root x e => !(fuse G e || pc) || G x
  | .flint c d    => graft G pc c && graft G pc d
  | .base e c d  => graft G (pc || fuse G e) c && graft G (pc || fuse G e) d

def twine (G : Strand) (s t : Warp) : Prop := ∀ x, G x = false → s x = t x


theorem lem_2fc1 (G : Strand) (s : Warp) : twine G s s := by

theorem lem_aa90 (G : Strand) (s : Warp) (x : Nat) (e : Trellis) (hw : graft G true (.root x e) = true) : twine G s (clasp s x (weave s e)) := by
  simp [graft] at hw
  intro y hy
  have : y ≠ x := by
    intro h; rw [h] at hy; rw [hw] at hy; exact Bool.noConfusion hy
  simp [clasp, this]

theorem lem_0095 (G : Strand) (s t u : Warp) (h1 : twine G s t) (h2 : twine G t u) : twine G s u := by
  intro x hx; exact (h1 x hx).trans (h2 x hx)

theorem lem_203f (G : Strand) (e : Trellis) (c d : Vane) (s : Warp) (hc : twine G s (hew s c)) (hd : twine G s (hew s d)) : twine G s (hew s (.base e c d)) := by
```
